<script>
    // 앱 실행 시 오늘 날짜를 기준으로 자동 조회
    window.onload = function() {
        const today = new Date().toISOString().split('T')[0];
        const dateInput = document.getElementById('meal-date');
        dateInput.value = today;
        fetchMealData();
    };

    function fetchMealData() {
        const dateInput = document.getElementById('meal-date').value;
        const mealContainer = document.getElementById('meal-container');

        if (!dateInput) return;

        // YYYY-MM-DD -> YYYYMMDD 변환
        const ymd = dateInput.replace(/-/g, '');

        // 변경된 경기도교육청(J10) 및 학교코드(7530469) 적용
        const officeCode = "J10";
        const schoolCode = "7530469";
        const apiUrl = `https://open.neis.go.kr/hub/mealServiceDietInfo?ATPT_OFCDC_SC_CODE=${officeCode}&SD_SCHUL_CODE=${schoolCode}&MLSV_YMD=${ymd}`;

        // 로딩 바 표시
        mealContainer.innerHTML = '<p class="info-msg loading">급식 정보를 불러오는 중입니다...</p>';

        fetch(apiUrl)
            .then(response => response.text()) // XML 파싱을 위해 텍스트로 수신
            .then(xmlString => {
                const parser = new DOMParser();
                const xmlDoc = parser.parseFromString(xmlString, "text/xml");
                
                // 에러 코드 검증 (<CODE> 태그 추출)
                const errorCodeNode = xmlDoc.getElementsByTagName("CODE")[0];
                const isError = errorCodeNode && errorCodeNode.textContent !== "INFO-000";

                const rows = xmlDoc.getElementsByTagName("row");

                if (rows.length > 0 && !isError) {
                    mealContainer.innerHTML = ''; // 화면 초기화

                    // 여러 끼니가 나올 수 있으므로 노드 순회
                    for (let i = 0; i < rows.length; i++) {
                        const row = rows[i];
                        
                        const mealType = row.getElementsByTagName("MMEAL_SC_NM")[0].textContent;
                        const calInfo = row.getElementsByTagName("CAL_INFO")[0].textContent;
                        let ddishNm = row.getElementsByTagName("DDISH_NM")[0].textContent;

                        // <br/> 태그 정제 및 가독성을 위한 불릿 추가
                        ddishNm = ddishNm.replace(/<br\/>/g, '<br>• ');

                        const card = document.createElement('div');
                        card.className = 'meal-card';
                        card.innerHTML = `
                            <div class="meal-type">${mealType} (${calInfo})</div>
                            <div class="meal-menu">• ${ddishNm}</div>
                        `;
                        mealContainer.appendChild(card);
                    }
                } else {
                    // 주말, 방학 등 급식이 없을 때
                    mealContainer.innerHTML = '<p class="info-msg">해당 날짜에는 급식 정보가 없습니다. <br><span>(주말, 공휴일 또는 방학)</span></p>';
                }
            })
            .catch(error => {
                console.error('Error:', error);
                mealContainer.innerHTML = '<p class="info-msg" style="color: #e74c3c;">데이터를 가져오는 중 오류가 발생했습니다.</p>';
            });
    }
</script>
